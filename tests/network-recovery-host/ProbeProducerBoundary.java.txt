  static JSONObject producerScenario(String name)throws Exception{
    android.os.SystemClock.offset=0;NativeNetwork n=new NativeNetwork(name);Process owned=n.process;boolean completed=false;String failure="";boolean passed=false;
    ProbeResult strict=null;
    try {
      if(name.startsWith("strict-")){
        String proxy=name.equals("strict-current")||name.equals("strict-stable")?"FixedExit":"ProbeExit1";
        String entry=proxy.equals("FixedExit")?"A":"B";Future<ProbeResult> future=n.probes.submit(()->n.probeMeasured(proxy,entry));
        if(!n.strictEntered.await(2,TimeUnit.SECONDS))throw new IOException("strict probe did not start");
        if(!name.equals("strict-stable"))n.invalidateVerifiedRoute();n.strictRelease.countDown();strict=future.get(2,TimeUnit.SECONDS);completed=true;
        passed=name.equals("strict-stable")?strict.delay==20&&strict.trace!=null&&strict.trace.comparable()&&n.probeCurrent(strict):strict.delay<0&&!n.probeCurrent(strict)&&n.verifications==0;
      }else{
        n.recoverExistingRoute();completed=true;
      }
    }catch(Exception expected){failure=expected.getClass().getSimpleName();}
    finally{Thread.interrupted();}
    boolean ownedAlive=n.transport==null?owned.isAlive():n.transport.alive("service-A");
    boolean recovered=completed&&n.currentEntry.equals("C")&&n.controllerEntry.equals("C")&&n.quality.selected.equals("C")&&n.verifiedRouteCurrent()&&ownedAlive&&n.verifications==2&&!n.trace.contains("verify:A")&&n.lastCandidate!=null&&n.lastCandidate.delay==20&&n.lastCandidate.trace!=null&&n.lastCandidate.trace.comparable()&&n.lastCandidate.binding.route<n.routeSerial;
    if(name.equals("ordinary")||name.equals("service-ordinary")||name.equals("producer-selection-window"))passed=recovered&&!n.probeCurrent(n.lastCandidate);
    else if(!name.startsWith("strict-")){
      passed=!completed&&!ownedAlive&&!n.ready&&n.verifiedRoute==null&&!n.trace.contains("verify:C")&&n.quality.selected.equals("A");
      if(name.equals("producer-replacement"))passed&=n.process!=owned&&n.process.isAlive();

    }
    JSONObject result=J.obj("name",name,"pass",passed,"completed",completed,"failure",failure,"trace",new JSONArray(n.trace),"producerObservations",n.producerObservations,"verificationCount",n.verifications,"selected",n.currentEntry,"committed",n.quality.selected,"ownedCoreAlive",ownedAlive,"ready",n.ready,"receiptPresent",n.verifiedRoute!=null,"finalProbeScopeRejected",n.lastCandidate==null||!n.probeCurrent(n.lastCandidate),"strictDelay",strict==null?0:strict.delay,"controlledClockAdvanceMs",android.os.SystemClock.offset);

  }
